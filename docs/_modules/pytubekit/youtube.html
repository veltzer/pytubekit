def youtube_dl_download_urls(urls: list[str]) -> None:
    logger = logging.getLogger(LOGGER_NAME)
    ydl_opts = {
        "extract_flat": True,
        "usenetrc": True,
    }
    logger.debug(f"passing options {ydl_opts}")
    with yt_dlp.YoutubeDL(ydl_opts) as ydl:
        ydl.download(urls)
